/* RFR_Golden — 外设初始化
 * 目标: CH32L103C8T6 / LQFP48 / QingKe V4C · RV32IMAFC
 * 时钟: SYSCLK 96 MHz, HCLK 96 MHz, PCLK1 96 MHz, PCLK2 96 MHz
 */
#include "rfr_init.h"

/* ───────────────── 系统时钟 ───────────────── */
void RFR_ClockInit(void)
{
    /* PLL 输入 3–25 MHz，输出 18–96 MHz（表3-16） */
    RCC_DeInit();
    RCC_HSICmd(ENABLE);
    while (RCC_GetFlagStatus(RCC_FLAG_HSIRDY) == RESET) { }

    EXTEN->EXTEN_CTR |= EXTEN_PLL_HSI_PRE;
    RCC_PLLConfig(RCC_PLLSource_HSI_Div2, RCC_PLLMul_12);
    RCC_PLLCmd(ENABLE);
    while (RCC_GetFlagStatus(RCC_FLAG_PLLRDY) == RESET) { }

    RCC_HCLKConfig(RCC_SYSCLK_Div1);
    RCC_PCLK2Config(RCC_HCLK_Div1);
    RCC_ADCCLKConfig(RCC_PCLK2_Div8);

    RCC_SYSCLKConfig(RCC_SYSCLKSource_PLLCLK);
    while (RCC_GetSYSCLKSource() != 0x08) { }

    /* Flash 等待周期由主频推导：96 MHz → 2 */
    FLASH_SetLatency(FLASH_Latency_2);
}

/* ───────────────── USART3 ───────────────── */
void RFR_USART3_Init(void)
{
    GPIO_InitTypeDef gpio = {0};
    USART_InitTypeDef uart = {0};
    RCC_PB1PeriphClockCmd(RCC_PB1Periph_USART3, ENABLE);
    /* BRR = 0x341 → 实际 115246.1 bit/s，误差 0.04% */
    uart.USART_BaudRate = 115200;
    uart.USART_WordLength = USART_WordLength_8b;
    uart.USART_StopBits = USART_StopBits_1;
    uart.USART_Parity = USART_Parity_No;
    uart.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    uart.USART_Mode = USART_Mode_Tx | USART_Mode_Rx;
    USART_Init(USART3, &uart);
    /* 同步模式：CK 输出波特率时钟（CTLR2.CLKEN=1） */
    USART_ClockInitTypeDef uartCk = {0};
    USART_ClockStructInit(&uartCk);
    uartCk.USART_Clock = USART_Clock_Enable;
    uartCk.USART_CPOL = USART_CPOL_Low;
    uartCk.USART_CPHA = USART_CPHA_1Edge;
    uartCk.USART_LastBit = USART_LastBit_Disable;
    USART_ClockInit(USART3, &uartCk);
    USART_Cmd(USART3, ENABLE);
    USART_ITConfig(USART3, USART_IT_RXNE, ENABLE);
}
