/* RFR_Golden — 外设初始化
 * 目标: CH32V467KEU6 / QFN32 / QingKe V3V · RV32IMAFCXV
 * 时钟: SYSCLK 200 MHz, HCLK 200 MHz, PCLK1 200 MHz, PCLK2 200 MHz
 */
#include "rfr_init.h"

/* ───────────────── 系统时钟 ───────────────── */
void RFR_ClockInit(void)
{
    /* HSI 8 MHz → PLL → 最高 200 MHz（DS 表1-1） */
    RCC_DeInit();
    RCC_HSICmd(ENABLE);
    while (RCC_GetFlagStatus(RCC_FLAG_HSIRDY) == RESET) { }

    RCC_PLLConfig(RCC_PLLSource_HSI, RCC_PLLMul_24);
    RCC_PLLCmd(ENABLE);
    while (RCC_GetFlagStatus(RCC_FLAG_PLLRDY) == RESET) { }

    RCC_HCLKConfig(RCC_SYSCLK_Div1);
    RCC_PCLK2Config(RCC_HCLK_Div2);
    RCC_ADCCLKConfig(RCC_PCLK2_Div8);

    RCC_SYSCLKConfig(RCC_SYSCLKSource_PLLCLK);
    while (RCC_GetSYSCLKSource() != 0x08) { }

}

1. 把导出的整个文件夹放到 WCH 官方 EVT 的 `EXAM/` 目录下（和 `GPIO_Toggle` 等例程平级），
   目录名保持 `RFR_Golden`。
2. MRS → File → Import → Existing Projects into Workspace，选中该文件夹。
   `Core` / `Startup` / `Ld` / `Peripheral` / `Debug` 五个目录通过
   linkedResources 指向 `../SRC/` 下的公共代码，不需要拷贝。
3. 直接 Build。产物在 `obj/RFR_Golden.hex`。

> `.wvproj` 是 WCH 加密的调试器配置文件，MRS 在首次保存工程时会自动生成，导出时不包含。
