/* RFR_Golden — 外设初始化
 * 目标: CH32V467KEU6 / QFN32 / QingKe V3V · RV32IMAFCXV
 * 时钟: SYSCLK 200 MHz, HCLK 200 MHz, PCLK1 200 MHz, PCLK2 200 MHz
 */
#include "rfr_init.h"

/* ───────────────── 系统时钟 ───────────────── */
void RFR_ClockInit(void)
{
    /* HSI 8 MHz → PLL → 最高 200 MHz（DS 表1-1） */
    RCC_DeInit();
    RCC_HSICmd(ENABLE);
    while (RCC_GetFlagStatus(RCC_FLAG_HSIRDY) == RESET) { }

    RCC_PLLConfig(RCC_PLLSource_HSI, RCC_PLLMul_24);
    RCC_PLLCmd(ENABLE);
    while (RCC_GetFlagStatus(RCC_FLAG_PLLRDY) == RESET) { }

    RCC_HCLKConfig(RCC_SYSCLK_Div1);
    RCC_PCLK2Config(RCC_HCLK_Div2);
    RCC_ADCCLKConfig(RCC_PCLK2_Div8);

    RCC_SYSCLKConfig(RCC_SYSCLKSource_PLLCLK);
    while (RCC_GetSYSCLKSource() != 0x08) { }

}

/* ───────────────── SPI3 ───────────────── */
void RFR_SPI3_Init(void)
{
    GPIO_InitTypeDef gpio = {0};
    SPI_InitTypeDef spi = {0};
    RCC_PB1PeriphClockCmd(RCC_PB1Periph_SPI3, ENABLE);
    /* fSCK = 25.00 MHz */
    spi.SPI_Direction = SPI_Direction_1Line_Tx;
    spi.SPI_Mode = SPI_Mode_Slave;
    spi.SPI_DataSize = SPI_DataSize_8b;
    spi.SPI_CPOL = SPI_CPOL_Low;
    spi.SPI_CPHA = SPI_CPHA_1Edge;
    spi.SPI_NSS = SPI_NSS_Soft;
    spi.SPI_BaudRatePrescaler = SPI_BaudRatePrescaler_8;
    spi.SPI_FirstBit = SPI_FirstBit_MSB;
    SPI_Init(SPI3, &spi);
    SPI_Cmd(SPI3, ENABLE);
    SPI_I2S_ITConfig(SPI3, SPI_I2S_IT_RXNE, ENABLE);
}
